def contient_ville(adresse, ville):
    if pd.isna(ville) or pd.isna(adresse):
        return False
    return ville.upper() in adresse.upper()

df_adresses['VillePresenteDansAdresse'] = df_adresses.apply(
    lambda row: contient_ville(row['AdresseComplete'], row['Ville']),
    axis=1
)

# Cas à vérifier : pas de correspondance textuelle ou faible confiance
df_a_verifier = df_adresses[
    (df_adresses['VillePresenteDansAdresse'] == False) | (df_adresses['Confiance'] < 70)
]

df_a_verifier.to_csv("villes_a_verifier.csv", index=False)
